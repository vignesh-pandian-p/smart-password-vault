<?php
session_start();
include 'db.php';
include 'includes/crypto.php';

if (!isset($_SESSION['user_id'])) {
    header("Location: login.php");
    exit;
}

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $user_id = $_SESSION['user_id'];
    $site = trim($_POST['site_name'] ?? '');
    $username = trim($_POST['site_username'] ?? '');
    $password = $_POST['site_password'] ?? '';
    
    if (!empty($site) && !empty($password)) {
        $encrypted = encrypt_password($password);

        $stmt = $conn->prepare("INSERT INTO vault (user_id, site_name, site_username, encrypted_password) VALUES (?, ?, ?, ?)");
        $stmt->bind_param("isss", $user_id, $site, $username, $encrypted);

        if ($stmt->execute()) {
            $_SESSION['flash_message'] = "Credential saved successfully!";
            $_SESSION['flash_type'] = "success";
        } else {
            $_SESSION['flash_message'] = "Error saving credential.";
            $_SESSION['flash_type'] = "error";
        }
        $stmt->close();
    }
}

header("Location: dashboard.php");
exit;

<?php
session_start();
include 'db.php';
include 'includes/crypto.php';

if (!isset($_SESSION['user_id'])) {
    header("Location: login.php");
    exit;
}

$user_id = $_SESSION['user_id'];

// Flash messages
$flash_message = $_SESSION['flash_message'] ?? null;
$flash_type = $_SESSION['flash_type'] ?? 'success';
unset($_SESSION['flash_message'], $_SESSION['flash_type']);

// Fetch user's credentials
$stmt = $conn->prepare("SELECT id, site_name, site_username, encrypted_password, created_at FROM vault WHERE user_id = ? ORDER BY id DESC");
$stmt->bind_param("i", $user_id);
$stmt->execute();
$result = $stmt->get_result();

$vault_items = [];
$weakCount = 0;
$staleCount = 0;

while ($row = $result->fetch_assoc()) {
    $plain = decrypt_password($row['encrypted_password']);
    $isWeak = false;
    $isStale = false;

    if ($plain !== null) {
        $len = strlen($plain);
        $hasLower = preg_match('/[a-z]/', $plain);
        $hasUpper = preg_match('/[A-Z]/', $plain);
        $hasDigit = preg_match('/[0-9]/', $plain);
        $hasSymbol = preg_match('/[^a-zA-Z0-9]/', $plain);
        $typeCount = ($hasLower ? 1 : 0) + ($hasUpper ? 1 : 0) + ($hasDigit ? 1 : 0) + ($hasSymbol ? 1 : 0);

        if ($len < 8 || $typeCount < 2) {
            $isWeak = true;
            $weakCount++;
        }
    }

    $createdTime = strtotime($row['created_at']);
    $ageDays = floor((time() - $createdTime) / 86400);
    if ($ageDays >= 90) {
        $isStale = true;
        $staleCount++;
    }

    $row['is_weak'] = $isWeak;
    $row['is_stale'] = $isStale;
    $row['age_days'] = $ageDays;

    $vault_items[] = $row;
}
$stmt->close();
$conn->close();

$page_title = "Vault Dashboard";
$itemCount = count($vault_items);
include 'includes/header.php';

-- Create the database
CREATE DATABASE password_vault;

-- Use the database
USE password_vault;

-- Users table
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL
);

-- Vault table (encrypted passwords)
CREATE TABLE vault (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    site_name VARCHAR(100),
    site_username VARCHAR(100),
    encrypted_password TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

import os

folder_path = r"C:\xampp\htdocs\smart-password-vault"
output_file = "all_codes.txt"

text_extensions = ('.php', '.sql', '.txt', '.css', '.js', '.py')
skip_dirs = {'assets', '.git'}

with open(output_file, "w", encoding="utf-8") as out:
    # First write root files
    for filename in sorted(os.listdir(folder_path)):
        file_path = os.path.join(folder_path, filename)
        if os.path.isfile(file_path) and filename != output_file and filename.endswith(text_extensions):
            out.write(f"\n===== {filename} =====\n\n")
            with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
                out.write(f.read())
                out.write("\n\n")

    # Then write includes
    includes_dir = os.path.join(folder_path, "includes")
    if os.path.isdir(includes_dir):
        for filename in sorted(os.listdir(includes_dir)):
            file_path = os.path.join(includes_dir, filename)
            if os.path.isfile(file_path) and filename.endswith(text_extensions):
                rel_name = os.path.join("includes", filename)
                out.write(f"\n===== {rel_name} =====\n\n")
                with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
                    out.write(f.read())
                    out.write("\n\n")

print(f"✅ All code merged into {output_file}")

    $user_id = $_SESSION['user_id'];
    $vault_id = intval($_POST['vault_id'] ?? 0);
    $site = trim($_POST['site_name'] ?? '');
    $username = trim($_POST['site_username'] ?? '');
    $password = $_POST['site_password'] ?? '';

    if ($vault_id > 0 && !empty($site)) {
        if (!empty($password)) {
            $encrypted = encrypt_password($password);
            $stmt = $conn->prepare("UPDATE vault SET site_name = ?, site_username = ?, encrypted_password = ? WHERE id = ? AND user_id = ?");
            $stmt->bind_param("sssii", $site, $username, $encrypted, $vault_id, $user_id);
        } else {
            $stmt = $conn->prepare("UPDATE vault SET site_name = ?, site_username = ? WHERE id = ? AND user_id = ?");
            $stmt->bind_param("ssii", $site, $username, $vault_id, $user_id);
        }

        if ($stmt->execute()) {
            $_SESSION['flash_message'] = "Credential updated successfully!";
            $_SESSION['flash_type'] = "success";
        } else {
            $_SESSION['flash_message'] = "Error updating credential.";
            $_SESSION['flash_type'] = "error";
        }
        $stmt->close();
    }
}

header("Location: dashboard.php");
exit;



===== includes\crypto.php =====

<?php
/**
 * Smart Password Vault - Cryptographic Utilities
 */

if (!defined('VAULT_CIPHER_KEY')) {
    define('VAULT_CIPHER_KEY', 'my_secret_encryption_key_123');
}

if (!defined('VAULT_CIPHER_IV')) {
    define('VAULT_CIPHER_IV', '1234567890123456');
}

/**
 * Encrypt a plain-text password using AES-256-CBC
 */
function encrypt_password($password) {
    return openssl_encrypt($password, "AES-256-CBC", VAULT_CIPHER_KEY, 0, VAULT_CIPHER_IV);
}

/**
 * Decrypt a base64-encoded ciphertext using AES-256-CBC
 */
function decrypt_password($encrypted) {
    $decoded = base64_decode($encrypted, true);
    if ($decoded === false) {
        return null;
    }
    $decrypted = openssl_decrypt($decoded, "AES-256-CBC", VAULT_CIPHER_KEY, OPENSSL_RAW_DATA, VAULT_CIPHER_IV);
    return ($decrypted !== false) ? $decrypted : null;
}
